   106→   106→/**
   107→   107→ * Generic handler that uses token optimizer for lazy loading
   108→   108→ */
   109→   109→function createOptimizedHandler(
   110→   110→  skillId: string,
   111→   111→  handlerGenerator: () => CommandHandler
   112→   112→): CommandHandler {
   113→   113→  return async (invocation) => {
   114→   114→    const optimizer = getTokenOptimizer()
   115→   115→    
   116→   116→    // Load skill on demand (token optimized!)
   117→   117→    await optimizer.loadSkill(skillId)
   118→   118→    
   119→   119→    // Generate and execute handler
   120→   120→    const handler = handlerGenerator()
   121→   121→    const startTime = Date.now()
   122→   122→    
   123→   123→    try {
   124→   124→      const result = await handler(invocation)
   125→   125→      
   126→   126→      // Record usage for learning
   127→   127→      const duration = Date.now() - startTime
   128→   128→      optimizer.recordUsage(skillId, duration)
   129→   129→      
   130→   130→      return result
   131→   131→    } catch (error) {
   132→   132→      return {
   133→   133→        kind: 'error' as const,
   134→   134→        text: `Error executing ${skillId}: ${error instanceof Error ? error.message : 'Unknown error'}`
   135→   135→      }
   136→   136→    }
   137→   137→  }
   138→   138→}
